# 变量替换使用案例

## 变量替换基础

FastD\Config 的变量替换功能允许在配置文件中使用占位符，并在解析时用实际值替换。变量使用 `%` 符号标记，格式为 `%variable_name%`。

### 基础变量替换示例

#### 配置文件 (config.yml)
```yaml
database:
  host: "%db_host%"
  port: "%db_port%"
  name: "%db_name%"
  username: "%db_username%"
  password: "%db_password%"
  url: "mysql://%db_username%:%db_password%@%db_host%:%db_port%/%db_name%"
```

#### 变量配置
```php
$variables = [
    'db_host' => 'localhost',
    'db_port' => 3306,
    'db_name' => 'myapp',
    'db_username' => 'root',
    'db_password' => 'password123'
];
```

#### 使用示例
```php
<?php
require_once 'vendor/autoload.php';

use FastD\Config\FileParser;

$variables = [
    'db_host' => 'localhost',
    'db_port' => 3306,
    'db_name' => 'myapp',
    'db_username' => 'root',
    'db_password' => 'password123'
];

$parser = new FileParser($variables);
$config = $parser->parse('config.yml');

// 验证变量替换结果
echo $config->get('database.host') . "\n";      // localhost
echo $config->get('database.port') . "\n";      // 3306
echo $config->get('database.url') . "\n";       // mysql://root:password123@localhost:3306/myapp
```

## 环境变量集成

### 从环境变量加载变量配置

```php
<?php
use FastD\Config\FileParser;

// 从环境变量构建变量配置
$variables = [
    'db_host' => $_ENV['DB_HOST'] ?? $_SERVER['DB_HOST'] ?? 'localhost',
    'db_port' => (int)($_ENV['DB_PORT'] ?? $_SERVER['DB_PORT'] ?? 3306),
    'db_name' => $_ENV['DB_NAME'] ?? $_SERVER['DB_NAME'] ?? 'myapp',
    'db_username' => $_ENV['DB_USERNAME'] ?? $_SERVER['DB_USERNAME'] ?? 'root',
    'db_password' => $_ENV['DB_PASSWORD'] ?? $_SERVER['DB_PASSWORD'] ?? 'password',
    'redis_host' => $_ENV['REDIS_HOST'] ?? 'localhost',
    'redis_port' => (int)($_ENV['REDIS_PORT'] ?? 6379),
    'app_env' => $_ENV['APP_ENV'] ?? 'development'
];

$parser = new FileParser($variables);
$config = $parser->parse('config/app.yml');

echo $config->get('database.host') . "\n";  // 来自环境变量或默认值
```

### 使用 .env 文件

```php
<?php
// 加载 .env 文件
if (file_exists('.env')) {
    $lines = file('.env', FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES);
    foreach ($lines as $line) {
        if (strpos($line, '=') !== false && strpos($line, '#') !== 0) {
            list($key, $value) = explode('=', $line, 2);
            $key = trim($key);
            $value = trim($value);
            // 移除引号
            $value = trim($value, '"\'');
            $_ENV[$key] = $value;
        }
    }
}

use FastD\Config\FileParser;

$variables = [
    'app_name' => $_ENV['APP_NAME'] ?? 'MyApp',
    'app_env' => $_ENV['APP_ENV'] ?? 'development',
    'database_url' => $_ENV['DATABASE_URL'] ?? 'sqlite::memory:',
    'redis_url' => $_ENV['REDIS_URL'] ?? 'redis://localhost:6379',
    'api_key' => $_ENV['API_KEY'] ?? '',
    'log_level' => $_ENV['LOG_LEVEL'] ?? 'info'
];

$parser = new FileParser($variables);
$config = $parser->parse('config/services.yml');
```

## 多环境配置管理

### 环境特定变量配置

```php
<?php
use FastD\Config\FileParser;

class EnvironmentConfig
{
    private $variables = [];
    
    public function __construct(string $environment = null)
    {
        $environment = $environment ?: ($_ENV['APP_ENV'] ?? 'development');
        $this->loadEnvironmentVariables($environment);
    }
    
    private function loadEnvironmentVariables(string $environment)
    {
        // 基础变量（所有环境共享）
        $baseVariables = [
            'app_name' => 'MyApplication',
            'app_version' => '1.0.0',
            'timezone' => 'Asia/Shanghai'
        ];
        
        // 环境特定变量
        $envVariables = match($environment) {
            'production' => [
                'db_host' => 'prod-db.example.com',
                'db_port' => 3306,
                'db_name' => 'prod_db',
                'db_username' => $_ENV['PROD_DB_USER'] ?? 'prod_user',
                'db_password' => $_ENV['PROD_DB_PASS'] ?? 'prod_pass',
                'redis_host' => 'prod-redis.example.com',
                'cache_driver' => 'redis',
                'log_level' => 'error',
                'debug' => false
            ],
            'staging' => [
                'db_host' => 'staging-db.example.com',
                'db_port' => 3306,
                'db_name' => 'staging_db',
                'db_username' => $_ENV['STAGING_DB_USER'] ?? 'staging_user',
                'db_password' => $_ENV['STAGING_DB_PASS'] ?? 'staging_pass',
                'redis_host' => 'staging-redis.example.com',
                'cache_driver' => 'redis',
                'log_level' => 'warning',
                'debug' => false
            ],
            default => [  // development
                'db_host' => 'localhost',
                'db_port' => 3306,
                'db_name' => 'dev_db',
                'db_username' => 'root',
                'db_password' => 'root',
                'redis_host' => 'localhost',
                'cache_driver' => 'file',
                'log_level' => 'debug',
                'debug' => true
            ]
        };
        
        $this->variables = array_merge($baseVariables, $envVariables);
    }
    
    public function getParser(): FileParser
    {
        return new FileParser($this->variables);
    }
    
    public function getVariables(): array
    {
        return $this->variables;
    }
}

// 使用示例
$envConfig = new EnvironmentConfig('production');
$parser = $envConfig->getParser();
$config = $parser->parse('config/app.yml');

echo $config->get('database.host') . "\n";  // 根据环境变化
echo $config->get('log_level') . "\n";      // 根据环境变化
```

### 配置文件模板

#### config/template.yml
```yaml
app:
  name: "%app_name%"
  version: "%app_version%"
  environment: "%app_env%"
  debug: "%debug%"

database:
  host: "%db_host%"
  port: "%db_port%"
  name: "%db_name%"
  username: "%db_username%"
  password: "%db_password%"
  driver: mysql
  charset: utf8mb4
  collation: utf8mb4_unicode_ci
  url: "mysql://%db_username%:%db_password%@%db_host%:%db_port%/%db_name%"

cache:
  driver: "%cache_driver%"
  host: "%redis_host%"
  port: "%redis_port%"
  prefix: "%app_name%_cache_"

logging:
  level: "%log_level%"
  path: "/var/log/%app_name%/%app_env%.log"
  max_files: 30

api:
  base_url: "https://api.%app_env%.example.com"
  key: "%api_key%"
  timeout: 30
```

## 复杂变量替换场景

### 嵌套变量

```php
<?php
use FastD\Config\FileParser;

// config/nested.yml
$nestedConfig = [
    'database' => [
        'host' => '%db_host%',
        'port' => '%db_port%',
        'name' => '%db_name%',
        'credentials' => [
            'username' => '%db_username%',
            'password' => '%db_password%'
        ]
    ],
    'database_url' => 'mysql://%db_username%:%db_password%@%db_host%:%db_port%/%db_name%'
];

// 变量配置
$variables = [
    'db_host' => 'localhost',
    'db_port' => 3306,
    'db_name' => 'myapp',
    'db_username' => 'root',
    'db_password' => 'password'
];

$parser = new FileParser($variables);
$config = $parser->parse('config/nested.yml');

// 所有变量都被正确替换
echo $config->get('database.host') . "\n";      // localhost
echo $config->get('database.credentials.username') . "\n";  // root
echo $config->get('database_url') . "\n";       // mysql://root:password@localhost:3306/myapp
```

### 条件变量配置

```php
<?php
use FastD\Config\FileParser;

class ConditionalVariables
{
    public static function getVariables(): array
    {
        $variables = [
            'app_env' => $_ENV['APP_ENV'] ?? 'development',
            'timezone' => $_ENV['TIMEZONE'] ?? 'Asia/Shanghai'
        ];
        
        // 根据环境设置不同的数据库配置
        $dbConfig = match($variables['app_env']) {
            'production' => [
                'db_host' => $_ENV['PROD_DB_HOST'] ?? 'prod-db.example.com',
                'db_port' => (int)($_ENV['PROD_DB_PORT'] ?? 3306),
                'db_ssl' => true
            ],
            'staging' => [
                'db_host' => $_ENV['STAGE_DB_HOST'] ?? 'stage-db.example.com',
                'db_port' => (int)($_ENV['STAGE_DB_PORT'] ?? 3306),
                'db_ssl' => true
            ],
            default => [
                'db_host' => $_ENV['DEV_DB_HOST'] ?? 'localhost',
                'db_port' => (int)($_ENV['DEV_DB_PORT'] ?? 3306),
                'db_ssl' => false
            ]
        };
        
        return array_merge($variables, $dbConfig);
    }
}

// 使用条件变量
$variables = ConditionalVariables::getVariables();
$parser = new FileParser($variables);
$config = $parser->parse('config/conditional.yml');
```

## 高级变量替换技巧

### 动态变量解析

```php
<?php
use FastD\Config\FileParser;

class DynamicVariables
{
    public static function resolveVariables(array $config): array
    {
        $resolved = [];
        
        // 首先解析基础变量
        $baseVariables = [
            'hostname' => gethostname(),
            'ip_address' => $_SERVER['SERVER_ADDR'] ?? '127.0.0.1',
            'current_time' => date('Y-m-d H:i:s'),
            'app_path' => dirname(__DIR__),
            'temp_dir' => sys_get_temp_dir()
        ];
        
        // 合并环境变量
        $envVariables = [];
        foreach ($_ENV as $key => $value) {
            if (preg_match('/^(DB_|REDIS_|API_|APP_)/', $key)) {
                $envVariables[strtolower($key)] = $value;
            }
        }
        
        return array_merge($baseVariables, $envVariables, $config);
    }
}

// 使用动态变量解析
$config = [
    'app' => [
        'server' => '%hostname%',
        'ip' => '%ip_address%',
        'start_time' => '%current_time%'
    ],
    'paths' => [
        'app' => '%app_path%',
        'temp' => '%temp_dir%'
    ]
];

$variables = DynamicVariables::resolveVariables($config);
$parser = new FileParser($variables);
$parsedConfig = $parser->parse('config/dynamic.yml');
```

### 变量验证和默认值

```php
<?php
use FastD\Config\FileParser;

class SafeVariables
{
    private array $required = [
        'db_host',
        'db_name', 
        'db_username'
    ];
    
    private array $defaults = [
        'db_port' => 3306,
        'db_charset' => 'utf8mb4',
        'log_level' => 'info',
        'debug' => false
    ];
    
    public function processVariables(array $variables): array
    {
        // 检查必需变量
        foreach ($this->required as $required) {
            if (empty($variables[$required])) {
                throw new InvalidArgumentException("必需的变量 {$required} 未设置");
            }
        }
        
        // 设置默认值
        foreach ($this->defaults as $key => $default) {
            if (!isset($variables[$key])) {
                $variables[$key] = $default;
            }
        }
        
        // 验证数据类型
        if (isset($variables['db_port'])) {
            $variables['db_port'] = (int)$variables['db_port'];
        }
        
        if (isset($variables['debug'])) {
            $variables['debug'] = filter_var($variables['debug'], FILTER_VALIDATE_BOOLEAN);
        }
        
        return $variables;
    }
}

// 使用安全变量处理
$rawVariables = [
    'db_host' => $_ENV['DB_HOST'] ?? 'localhost',
    'db_name' => $_ENV['DB_NAME'] ?? 'myapp',
    'db_username' => $_ENV['DB_USERNAME'] ?? 'root',
    'db_password' => $_ENV['DB_PASSWORD'] ?? 'password'
];

$safeVariables = new SafeVariables();
$processedVariables = $safeVariables->processVariables($rawVariables);

$parser = new FileParser($processedVariables);
$config = $parser->parse('config/safe.yml');
```

## 实际应用案例

### Web 应用配置

```php
<?php
use FastD\Config\FileParser;

class WebAppConfig
{
    public function __construct()
    {
        // 从多种来源收集变量
        $variables = $this->collectVariables();
        
        $this->parser = new FileParser($variables);
    }
    
    private function collectVariables(): array
    {
        return [
            // 应用基本信息
            'app_name' => $_ENV['APP_NAME'] ?? 'WebApp',
            'app_env' => $_ENV['APP_ENV'] ?? 'development',
            'app_debug' => filter_var($_ENV['APP_DEBUG'] ?? 'false', FILTER_VALIDATE_BOOLEAN),
            'app_url' => $_ENV['APP_URL'] ?? 'http://localhost',
            
            // 数据库配置
            'db_connection' => $_ENV['DB_CONNECTION'] ?? 'mysql',
            'db_host' => $_ENV['DB_HOST'] ?? '127.0.0.1',
            'db_port' => (int)($_ENV['DB_PORT'] ?? 3306),
            'db_database' => $_ENV['DB_DATABASE'] ?? 'forge',
            'db_username' => $_ENV['DB_USERNAME'] ?? 'forge',
            'db_password' => $_ENV['DB_PASSWORD'] ?? '',
            
            // Redis 配置
            'redis_client' => $_ENV['REDIS_CLIENT'] ?? 'phpredis',
            'redis_host' => $_ENV['REDIS_HOST'] ?? '127.0.0.1',
            'redis_password' => $_ENV['REDIS_PASSWORD'] ?? null,
            'redis_port' => (int)($_ENV['REDIS_PORT'] ?? 6379),
            'redis_database' => (int)($_ENV['REDIS_DB'] ?? 0),
            
            // 邮件配置
            'mail_mailer' => $_ENV['MAIL_MAILER'] ?? 'smtp',
            'mail_host' => $_ENV['MAIL_HOST'] ?? 'smtp.mailgun.org',
            'mail_port' => (int)($_ENV['MAIL_PORT'] ?? 587),
            'mail_username' => $_ENV['MAIL_USERNAME'] ?? null,
            'mail_password' => $_ENV['MAIL_PASSWORD'] ?? null,
            'mail_encryption' => $_ENV['MAIL_ENCRYPTION'] ?? null,
            'mail_from_address' => $_ENV['MAIL_FROM_ADDRESS'] ?? 'hello@example.com',
            
            // 日志配置
            'log_level' => $_ENV['LOG_LEVEL'] ?? 'debug',
            'log_channel' => $_ENV['LOG_CHANNEL'] ?? 'stack',
            
            // 其他服务
            'queue_connection' => $_ENV['QUEUE_CONNECTION'] ?? 'sync',
            'cache_store' => $_ENV['CACHE_STORE'] ?? 'file'
        ];
    }
    
    public function loadConfig(string $configFile): \FastD\Config\Parsed
    {
        return $this->parser->parse($configFile);
    }
}

// 使用 Web 应用配置
$appConfig = new WebAppConfig();
$mainConfig = $appConfig->loadConfig('config/app.yml');
$databaseConfig = $appConfig->loadConfig('config/database.yml');
```

### 微服务配置

```php
<?php
use FastD\Config\FileParser;

class MicroServiceConfig
{
    private FileParser $parser;
    
    public function __construct(string $serviceName)
    {
        $variables = $this->getServiceVariables($serviceName);
        $this->parser = new FileParser($variables);
    }
    
    private function getServiceVariables(string $serviceName): array
    {
        return [
            // 服务基本信息
            'service_name' => $serviceName,
            'service_version' => $_ENV["{$serviceName}_VERSION"] ?? '1.0.0',
            'service_port' => (int)($_ENV["{$serviceName}_PORT"] ?? 8000),
            
            // 服务注册与发现
            'registry_host' => $_ENV['REGISTRY_HOST'] ?? 'localhost',
            'registry_port' => (int)($_ENV['REGISTRY_PORT'] ?? 8500),
            
            // 数据库配置
            'db_host' => $_ENV["{$serviceName}_DB_HOST"] ?? "${serviceName}-db",
            'db_port' => (int)($_ENV["{$serviceName}_DB_PORT"] ?? 3306),
            'db_name' => $_ENV["{$serviceName}_DB_NAME"] ?? $serviceName,
            'db_user' => $_ENV["{$serviceName}_DB_USER"] ?? 'root',
            'db_pass' => $_ENV["{$serviceName}_DB_PASS"] ?? 'password',
            
            // 缓存配置
            'cache_host' => $_ENV["{$serviceName}_CACHE_HOST"] ?? "${serviceName}-cache",
            'cache_port' => (int)($_ENV["{$serviceName}_CACHE_PORT"] ?? 6379),
            
            // 消息队列配置
            'queue_host' => $_ENV['QUEUE_HOST'] ?? 'localhost',
            'queue_port' => (int)($_ENV['QUEUE_PORT'] ?? 5672),
            'queue_user' => $_ENV['QUEUE_USER'] ?? 'guest',
            'queue_pass' => $_ENV['QUEUE_PASS'] ?? 'guest',
            
            // 其他服务依赖
            'user_service_url' => $_ENV['USER_SERVICE_URL'] ?? 'http://user-service:8001',
            'order_service_url' => $_ENV['ORDER_SERVICE_URL'] ?? 'http://order-service:8002',
            'payment_service_url' => $_ENV['PAYMENT_SERVICE_URL'] ?? 'http://payment-service:8003'
        ];
    }
    
    public function getConfig(string $configFile): \FastD\Config\Parsed
    {
        return $this->parser->parse($configFile);
    }
}

// 使用微服务配置
$authServiceConfig = new MicroServiceConfig('auth-service');
$authConfig = $authServiceConfig->getConfig('config/service.yml');
```
