# 单元测试说明和最佳实践

## 测试环境搭建

### 安装测试依赖

FastD\Config 使用 PHPUnit 进行单元测试。首先确保安装了开发依赖：

```bash
composer install --dev
```

或者单独安装 PHPUnit：

```bash
composer require --dev phpunit/phpunit:^9.0
```

### 配置 PHPUnit

项目根目录中的 `phpunit.xml` 文件定义了测试配置：

```xml
<?xml version="1.0" encoding="UTF-8"?>
<phpunit bootstrap="vendor/autoload.php"
         colors="true"
         convertErrorsToExceptions="true"
         convertNoticesToExceptions="true"
         convertWarningsToExceptions="true"
         processIsolation="false"
         stopOnFailure="false">
    <testsuites>
        <testsuite name="Application Test Suite">
            <directory>./tests/</directory>
        </testsuite>
    </testsuites>
    <coverage processUncoveredFiles="true">
        <include>
            <directory suffix=".php">./src/</directory>
        </include>
    </coverage>
</testsuites>
</phpunit>
```

### 运行测试

```bash
# 运行所有测试
./vendor/bin/phpunit

# 运行特定测试文件
./vendor/bin/phpunit tests/FileParserTest.php

# 运行特定测试方法
./vendor/bin/phpunit tests/FileParserTest.php --filter=testLoadJsonFile

# 生成代码覆盖率报告
./vendor/bin/phpunit --coverage-html coverage/
```

```php
use FastD\Config\FileParser;
use PHPUnit\Framework\TestCase;

class FileParserTest extends TestCase

    public function testConstructorWithArrayVariables()
    {
        $variables = [
            'database_host' => 'localhost',
            'database_port' => 3306
        ];
        
        $parser = new FileParser($variables);
        
        $this->assertInstanceOf(FileParser::class, $parser);
```

### 完整的 Parsed 类测试

```php
<?php
// tests/ParsedTest.php
use FastD\Config\Parsed;
use PHPUnit\Framework\TestCase;

class ParsedTest extends TestCase
{
    public function testConstructorWithArray()
    {
        $data = ['key' => 'value', 'nested' => ['subkey' => 'subvalue']];
        $parsed = new Parsed($data);
        
        $this->assertInstanceOf(Parsed::class, $parsed);
        $this->assertEquals('value', $parsed->get('key'));
        $this->assertEquals('subvalue', $parsed->get('nested.subkey'));
    }
    
    public function testGetExistingKey()
    {
        $parsed = new Parsed(['app' => 'MyApp', 'database' => ['host' => 'localhost']]);
        
        $this->assertEquals('MyApp', $parsed->get('app'));
        $this->assertEquals('localhost', $parsed->get('database.host'));
    }
    
    public function testGetNonExistingKeyWithDefault()
    {
        $parsed = new Parsed(['app' => 'MyApp']);
        
        $this->assertEquals('default_value', $parsed->get('non_existing', 'default_value'));
        $this->assertNull($parsed->get('another_non_existing'));
    }
    
    public function testGetWithDotInKeyName()
    {
        $parsed = new Parsed([
            'database.host' => 'direct_value',
            'database' => ['host' => 'nested_value']
        ]);
        
        // Direct key takes precedence
        $this->assertEquals('direct_value', $parsed->get('database.host'));
    }
    
    public function testSetSimpleValue()
    {
        $parsed = new Parsed();
        $parsed->set('app.name', 'MyApp');
        
        $this->assertEquals('MyApp', $parsed->get('app.name'));
    }
    
    public function testSetNestedValue()
    {
        $parsed = new Parsed();
        $parsed->set('database.host', 'localhost');
        $parsed->set('database.credentials.username', 'root');
        
        $this->assertEquals('localhost', $parsed->get('database.host'));
        $this->assertEquals('root', $parsed->get('database.credentials.username'));
    }
    
    public function testSetOverwritesExistingValue()
    {
        $parsed = new Parsed(['app' => 'OldApp']);
        $parsed->set('app', 'NewApp');
        
        $this->assertEquals('NewApp', $parsed->get('app'));
    }
    
    public function testHasExistingKey()
    {
        $parsed = new Parsed(['database' => ['host' => 'localhost']]);
        
        $this->assertTrue($parsed->has('database'));
        $this->assertTrue($parsed->has('database.host'));
        $this->assertFalse($parsed->has('database.port'));
    }
    
    public function testHasWithDotInKeyName()
    {
        $parsed = new Parsed([
            'database.host' => 'value1',
            'database' => ['host' => 'value2']
        ]);
        
        $this->assertTrue($parsed->has('database.host')); // Direct key exists
    }
    
    public function testUnsetSimpleKey()
    {
        $parsed = new Parsed(['app' => 'MyApp', 'database' => 'local']);
        $parsed->unset('app');
        
        $this->assertFalse($parsed->has('app'));
        $this->assertTrue($parsed->has('database'));
    }
    
    public function testUnsetNestedKey()
    {
        $parsed = new Parsed([
            'database' => [
                'host' => 'localhost',
                'port' => 3306,
                'credentials' => [
                    'username' => 'root',
                    'password' => 'password'
                ]
            ]
        ]);
        
        $parsed->unset('database.credentials.password');
        
        $this->assertTrue($parsed->has('database.credentials.username'));
        $this->assertFalse($parsed->has('database.credentials.password'));
        $this->assertTrue($parsed->has('database.host'));
    }
    
    public function testMergeSimpleArrays()
    {
        $parsed = new Parsed(['app' => 'MyApp']);
        $parsed->merge(['database' => 'mysql']);
        
        $this->assertEquals('MyApp', $parsed->get('app'));
        $this->assertEquals('mysql', $parsed->get('database'));
    }
    
    public function testMergeNestedArrays()
    {
        $parsed = new Parsed([
            'database' => [
                'host' => 'localhost',
                'port' => 3306
            ]
        ]);
        
        $parsed->merge([
            'database' => [
                'username' => 'root',
                'password' => 'password'
            ],
            'cache' => [
                'driver' => 'redis'
            ]
        ]);
        
        $this->assertEquals('localhost', $parsed->get('database.host'));
        $this->assertEquals(3306, $parsed->get('database.port'));
        $this->assertEquals('root', $parsed->get('database.username'));
        $this->assertEquals('password', $parsed->get('database.password'));
        $this->assertEquals('redis', $parsed->get('cache.driver'));
    }
    
    public function testMergeWithDifferentDataTypes()
    {
        $parsed = new Parsed([
            'database' => 'simple_value'
        ]);
        
        $parsed->merge([
            'database' => [
                'host' => 'localhost',
                'port' => 3306
            ]
        ]);
        
        // Original simple value should be converted to array and merged
        $this->assertEquals('localhost', $parsed->get('database.host'));
        $this->assertEquals(3306, $parsed->get('database.port'));
    }
    
    public function testArrayAccessOffsetGet()
    {
        $parsed = new Parsed(['app' => ['name' => 'MyApp']]);
        
        $this->assertEquals('MyApp', $parsed['app.name']);
    }
    
    public function testArrayAccessOffsetSet()
    {
        $parsed = new Parsed();
        $parsed['app.name'] = 'MyApp';
        
        $this->assertEquals('MyApp', $parsed->get('app.name'));
    }
    
    public function testArrayAccessOffsetExists()
    {
        $parsed = new Parsed(['app' => 'MyApp']);
        
        $this->assertTrue(isset($parsed['app']));
        $this->assertFalse(isset($parsed['nonexistent']));
    }
    
    public function testArrayAccessOffsetUnset()
    {
        $parsed = new Parsed(['app' => 'MyApp', 'database' => 'mysql']);
        unset($parsed['app']);
        
        $this->assertFalse(isset($parsed['app']));
        $this->assertTrue(isset($parsed['database']));
    }
    
    public function testIteration()
    {
        $data = ['app' => 'MyApp', 'version' => '1.0'];
        $parsed = new Parsed($data);
        
        $result = [];
        foreach ($parsed as $key => $value) {
            $result[$key] = $value;
        }
        
        $this->assertEquals($data, $result);
    }
    
    public function testCount()
    {
        $parsed = new Parsed(['app' => 'MyApp', 'version' => '1.0']);
        $this->assertCount(2, $parsed);
    }
    
    public function testComplexNestedOperations()
    {
        $parsed = new Parsed();
        
        // Set deeply nested values
        $parsed->set('services.database.host', 'db1.example.com');
        $parsed->set('services.database.slaves.0.host', 'slave1.example.com');
        $parsed->set('services.database.slaves.1.host', 'slave2.example.com');
        $parsed->set('services.cache.nodes.0', 'cache1.example.com');
        $parsed->set('services.cache.nodes.1', 'cache2.example.com');
        
        // Verify values
        $this->assertEquals('db1.example.com', $parsed->get('services.database.host'));
        $this->assertEquals('slave1.example.com', $parsed->get('services.database.slaves.0.host'));
        $this->assertEquals('slave2.example.com', $parsed->get('services.database.slaves.1.host'));
        $this->assertEquals('cache1.example.com', $parsed->get('services.cache.nodes.0'));
        $this->assertEquals('cache2.example.com', $parsed->get('services.cache.nodes.1'));
        
        // Update nested values
        $parsed->set('services.database.host', 'db2.example.com');
        $this->assertEquals('db2.example.com', $parsed->get('services.database.host'));
        
        // Remove nested values
        $parsed->unset('services.database.slaves.0.host');
        $this->assertFalse($parsed->has('services.database.slaves.0.host'));
        $this->assertTrue($parsed->has('services.database.slaves.1.host'));
    }
    
    public function testEdgeCases()
    {
        $parsed = new Parsed();
        
        // Empty key
        $this->assertNull($parsed->get(''));
        $this->assertFalse($parsed->has(''));
        
        // Null key
        $this->assertNull($parsed->get(null));
        $this->assertFalse($parsed->has(null));
        
        // Numeric keys
        $parsed->set('0', 'zero');
        $parsed->set('1.test', 'one-test');
        
        $this->assertEquals('zero', $parsed->get('0'));
        $this->assertEquals('one-test', $parsed->get('1.test'));
        
        // Special characters in keys
        $parsed->set('special.key_with_underscore', 'underscore_value');
        $parsed->set('special.key-with-dash', 'dash_value');
        
        $this->assertEquals('underscore_value', $parsed->get('special.key_with_underscore'));
        $this->assertEquals('dash_value', $parsed->get('special.key-with-dash'));
    }
}
```

### 3. 使用测试 Trait

```php
<?php
trait ConfigTestTrait
{
    private array $tempFiles = [];
    
    protected function setUp(): void
    {
        parent::setUp();
        $this->tempFiles = [];
    }
    
    protected function tearDown(): void
    {
        // Clean up temporary files
        foreach ($this->tempFiles as $file) {
            if (file_exists($file)) {
                unlink($file);
            }
        }
        parent::tearDown();
    }
    
    protected function createTempConfig(array $data, string $extension = 'json'): string
    {
        $filename = tempnam(sys_get_temp_dir(), 'test_config_') . ".{$extension}";
        $this->tempFiles[] = $filename;
        
        switch ($extension) {
            case 'json':
                file_put_contents($filename, json_encode($data));
                break;
            case 'yml':
            case 'yaml':
                file_put_contents($filename, yaml_emit($data));
                break;
            case 'ini':
                $this->writeIniFile($data, $filename);
                break;
            case 'php':
                file_put_contents($filename, '<?php return ' . var_export($data, true) . ';');
                break;
        }
        
        return $filename;
    }
    
    private function writeIniFile(array $data, string $filename): void
    {
        $content = '';
        foreach ($data as $section => $values) {
            if (is_array($values)) {
                $content .= "[{$section}]\n";
                foreach ($values as $key => $value) {
                    $content .= "{$key}={$value}\n";
                }
                $content .= "\n";
            } else {
                $content .= "{$section}={$values}\n";
            }
        }
        file_put_contents($filename, $content);
    }
}
```

```php
<?php
use PHPUnit\Framework\TestCase;

class ParameterizedConfigTest extends TestCase
{
    /**
     * @dataProvider configFormatProvider
     */
    public function testLoadDifferentFormats(string $format, array $data, string $expectedHost)
    {
        $filename = $this->createConfigFile($data, $format);
        
        $parser = new FileParser();
        $loadedData = $parser->load($filename);
        
        $this->assertEquals($expectedHost, $loadedData['database']['host'] ?? null);
        
        unlink($filename);
    }
    
    public function configFormatProvider(): array
    {
        $data = [
            'database' => [
                'host' => 'localhost',
                'port' => 3306
            ]
        ];
        
        return [
            'JSON format' => ['json', $data, 'localhost'],
            'YAML format' => ['yml', $data, 'localhost'],
            'PHP format' => ['php', $data, 'localhost'],
            'INI format' => ['ini', $data, 'localhost']
        ];
    }
    
    /**
     * @dataProvider variableReplacementProvider
     */
    public function testVariableReplacement(array $config, array $variables, string $expectedResult)
    {
        $parser = new FileParser($variables);
        
        $reflection = new ReflectionClass($parser);
        $method = $reflection->getMethod('replace');
        $method->setAccessible(true);
        
        $result = $method->invoke($parser, $config);
        
        $this->assertEquals($expectedResult, $result);
    }
    
    public function variableReplacementProvider(): array
    {
        return [
            'Simple replacement' => [
                '%name%',

                'John'

                'Hello John, welcome to our site'
            ],
```

```php
class IntegrationConfigTest extends TestCase
```

## 性能测试

```php
<?php
class PerformanceConfigTest extends TestCase
{
    public function testLargeConfigParsingPerformance()
    {
        // 创建大型配置文件
        $largeConfig = [];
        for ($i = 0; $i < 1000; $i++) {
            $largeConfig["section_{$i}"] = [
                'key1' => "value_{$i}",
                'key2' => $i,
                'nested' => [
                    'subkey' => "subvalue_{$i}"
                ]
            ];
        }
        
        $configFile = tempnam(sys_get_temp_dir(), 'large_config_');
        file_put_contents($configFile, json_encode($largeConfig));
        
        $startTime = microtime(true);
        
        $parser = new FileParser();
        $result = $parser->parse($configFile);
        
        $endTime = microtime(true);
        $duration = ($endTime - $startTime) * 1000; // Convert to milliseconds
        
        // Ensure parsing completes within 500ms
        $this->assertLessThan(500, $duration, "Large config parsing took too long: {$duration}ms");
        
        // Verify data integrity
        $this->assertEquals('value_0', $result->get('large_config.section_0.key1'));
        $this->assertEquals(999, $result->get('large_config.section_999.key2'));
        
        unlink($configFile);
    }
    
    public function testVariableReplacementPerformance()
    {
        $config = [];
        for ($i = 0; $i < 100; $i++) {
            $config["item_{$i}"] = "%variable_{$i}%";
        }
        
        $variables = [];
        for ($i = 0; $i < 100; $i++) {
            $variables["variable_{$i}"] = "replaced_value_{$i}";
        }
        
        $parser = new FileParser($variables);
        
        $startTime = microtime(true);
        
        $reflection = new ReflectionClass($parser);
        $method = $reflection->getMethod('replace');
        $method->setAccessible(true);
        
        $result = $method->invoke($parser, $config);
        
        $endTime = microtime(true);
        $duration = ($endTime - $startTime) * 1000;
        
        $this->assertLessThan(100, $duration, "Variable replacement took too long: {$duration}ms");
        
        $this->assertEquals('replaced_value_0', $result['item_0']);
        $this->assertEquals('replaced_value_99', $result['item_99']);
    }
}
```
